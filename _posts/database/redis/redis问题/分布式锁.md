package cache


import (
	"context"
	"strconv"

	"github.com/gomodule/redigo/redis"
	"gitlab.xiaoduoai.com/ecrobot/message-receiver/config"
	"gitlab.xiaoduoai.com/golib/xd_sdk/logger"
)

const (
	// dw平台缓存相关常量
	PrefixDWOrderId         = "dw:order_id:"
	DWOrderIncompleteKey    = "dw:order_incomplete:"
	DWClientMaxSyncCountKey = "dw:client_max_sync_count:"
	DWOrderMsgQueueKey      = "dw:order_msg_queue:"     // 订单消息队列key
	DWOrderIdToOpenUIDKey   = "dw:order_id_to_openuid:" // 订单ID到OpenUID的映射key
	DWOpenUIDToOneIDKey     = "dw:openuid_to_oneid:"    // OpenUID到OneID的映射key
	DWOrderIdToOneIDKey     = "dw:order_id_to_oneid:"   // 订单ID到OneID的映射key
)

// GetClientMaxOrderSyncCount 获取客户端最大订单同步数量
func GetClientMaxOrderSyncCount(ctx context.Context, mallId string, defaultCount int) (int, error) {
	key := DWClientMaxSyncCountKey + mallId
	reply, err := config.GetConfig().RedisConf.Redis.Client.DoWithContext(ctx, "GET", key)
	if err != nil {
		if err == redis.ErrNil {
			return defaultCount, nil
		}
		logger.Warningf(ctx, "get client max order sync count failed: %s", err.Error())
		return defaultCount, err
	}
	if reply == nil {
		return defaultCount, nil
	}
	count, err := strconv.Atoi(string(reply.([]byte)))
	if err != nil {
		logger.Warningf(ctx, "parse client max order sync count failed: %s", err.Error())
		return defaultCount, err
	}
	return count, nil
}

// SetClientMaxOrderSyncCount 设置客户端最大订单同步数量
func SetClientMaxOrderSyncCount(mallId string, count int) error {
	key := DWClientMaxSyncCountKey + mallId
	_, err := config.GetConfig().RedisConf.Redis.Client.DoWithContext(context.Background(), "SET", key, count, "EX", 60*60*24*7) // 7天过期
	return err
}

// 获取未完成的订单列表
func GetOrderInComplete(ctx context.Context, mallId string, limit int) ([]string, error) {
	key := DWOrderIncompleteKey + mallId
	reply, err := config.GetConfig().RedisConf.Redis.Client.DoWithContext(ctx, "LRANGE", key, 0, limit-1)
	if err != nil {
		logger.Warningf(ctx, "get order incomplete failed: %s", err.Error())
		return nil, err
	}
	if reply == nil {
		return []string{}, nil
	}

	var orders []string
	for _, item := range reply.([]interface{}) {
		orders = append(orders, string(item.([]byte)))
	}
	return orders, nil
}

// 保存未完成的订单
func SaveOrderInComplete(ctx context.Context, mallId string, orderId string) error {
	key := DWOrderIncompleteKey + mallId
	_, err := config.GetConfig().RedisConf.Redis.Client.DoWithContext(ctx, "RPUSH", key, orderId)
	if err != nil {
		logger.Warningf(ctx, "save order incomplete failed: %s", err.Error())
		return err
	}
	// 设置过期时间，防止数据堆积
	_, err = config.GetConfig().RedisConf.Redis.Client.DoWithContext(ctx, "EXPIRE", key, 60*60*24*15) // 15天过期
	return err
}

// 从未完成订单列表中移除订单
func RemoveOrderInComplete(ctx context.Context, mallId string, orderId string) error {
	key := DWOrderIncompleteKey + mallId
	_, err := config.GetConfig().RedisConf.Redis.Client.DoWithContext(ctx, "LREM", key, 0, orderId)
	if err != nil {
		logger.Warningf(ctx, "remove order incomplete failed: %s", err.Error())
		return err
	}
	return nil
}

// 安全地将订单消息添加到队列
func SafeAppendOrderMsgByOrderID(ctx context.Context, orderId, msg string) error {
	key := DWOrderMsgQueueKey + orderId
	_, err := config.GetConfig().RedisConf.Redis.Client.DoWithContext(ctx, "RPUSH", key, msg)
	if err != nil {
		logger.Warningf(ctx, "append order message failed: %s", err.Error())
		return err
	}
	// 设置过期时间，防止数据堆积
	_, err = config.GetConfig().RedisConf.Redis.Client.DoWithContext(ctx, "EXPIRE", key, 60*60*24*15) // 15天过期
	return err
}

// 安全地获取订单的所有消息
func SafeGetAllOrderMsgsByOrderID(ctx context.Context, orderId string) ([]string, error) {
	key := DWOrderMsgQueueKey + orderId
	reply, err := config.GetConfig().RedisConf.Redis.Client.DoWithContext(ctx, "LRANGE", key, 0, -1)
	if err != nil {
		if err == redis.ErrNil {
			return []string{}, nil
		}
		return nil, err
	}
	if reply == nil {
		return []string{}, nil
	}

	var msgs []string
	for _, item := range reply.([]interface{}) {
		msgs = append(msgs, string(item.([]byte)))
	}
	return msgs, nil
}

// 安全地删除订单的所有消息
func SafeDeleteAllOrderMsgsByOrderID(ctx context.Context, orderId string) error {
	key := DWOrderMsgQueueKey + orderId
	_, err := config.GetConfig().RedisConf.Redis.Client.DoWithContext(ctx, "DEL", key)
	return err
}

// 设置订单ID到OneID的映射
func SetOrderIdToOneId(ctx context.Context, orderId, oneId string) error {
	key := DWOrderIdToOneIDKey + orderId
	_, err := config.GetConfig().RedisConf.Redis.Client.DoWithContext(ctx, "SET", key, oneId, "EX", 60*60*24*15) // 15天过期
	return err
}

// 根据订单ID获取OneID
func GetOneIdByOrderId(ctx context.Context, orderId string) (string, error) {
	key := DWOrderIdToOneIDKey + orderId
	reply, err := config.GetConfig().RedisConf.Redis.Client.DoWithContext(ctx, "GET", key)
	if err != nil {
		if err == redis.ErrNil {
			return "", nil
		}
		return "", err
	}
	if reply == nil {
		return "", nil
	}
	return string(reply.([]byte)), nil
}
