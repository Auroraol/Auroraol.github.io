- 一个订单被锁住时，只会跳过这个订单（continue）

- 订单之间是并行的，互不影响

// WithDistributedLock 使用分布式锁执行操作的通用函数
// 这个函数封装了分布式锁的获取和释放逻辑，确保锁的正确释放
func WithDistributedLock(ctx context.Context, locker dlock.Locker, lockName string, operation func() error) error {
	if locker == nil {
		return errors.New("dlock client not initialized")
	}

	// 获取分布式锁
	unlock, err := locker.Lock(ctx, lockName)
	if err != nil {
		logger.Errorf(ctx, "Failed to acquire distributed lock: %v, lockName: %s", err, lockName)
		return errors.New("failed to acquire lock")
	}

	// 释放锁
	defer func() {
		ok, unlockErr := unlock()
		if !ok || unlockErr != nil {
			logger.Errorf(ctx, "Failed to release distributed lock: %v, lockName: %s, ok: %v", unlockErr, lockName, ok)
		}
	}()

	// 执行实际操作
	return operation()
}

// WithDistributedLockWithResult 使用分布式锁执行操作并返回结果的通用函数
// 这个函数适用于需要返回值的操作
